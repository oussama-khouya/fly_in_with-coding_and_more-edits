# Edge Case 19: Reverse Funnel Expansion
# Tests an inverted funnel topology: starting with a strict 1-drone needle chokepoint (neck, max_drones=1),
# which immediately bursts open into a wide delta fan of 4 parallel routes with varying speeds and zone types
# (priority, normal, restricted, and multi-hop detour) all reconverging at the goal.
# Verifies pipeline queueing at the start followed by rapid multi-channel fan-out.

nb_drones: 10

start_hub: start 0 0 [color=green]
hub: neck 1 0 [zone=normal color=orange max_drones=1]
hub: prio_route 2 2 [zone=priority color=cyan max_drones=1]
hub: norm_route 2 1 [zone=normal color=blue max_drones=2]
hub: restr_route 2 -1 [zone=restricted color=red max_drones=2]
hub: detour_a 2 -2 [zone=normal color=purple max_drones=3]
hub: detour_b 3 -2 [zone=normal color=purple max_drones=3]
end_hub: goal 4 0 [color=green]

# Ingress needle chokepoint
connection: start-neck [max_link_capacity=1]

# Fan-out routes from neck
