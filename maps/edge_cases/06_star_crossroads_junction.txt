# Edge Case 6: Star Crossroads Junction
# Tests multi-ingress and multi-egress through a central restricted roundabout.
# Drones launch from start into two independent approach corridors (North and South).
# Both approach corridors feed into 'roundabout', which has zone=restricted and max_drones=2.
# Once leaving the roundabout, drones can disburse via two exit corridors to reach goal.
# Tests concurrent restricted multi-drone reservation (up to 2 drones) from divergent directions.

nb_drones: 8

start_hub: start 0 0 [color=green]
hub: in_north 1 2 [zone=normal color=blue max_drones=2]
hub: in_south 1 -2 [zone=normal color=blue max_drones=2]
hub: roundabout 2 0 [zone=restricted color=red max_drones=2]
hub: out_north 3 2 [zone=normal color=cyan max_drones=2]
hub: out_south 3 -2 [zone=normal color=cyan max_drones=2]
end_hub: goal 4 0 [color=green]

connection: start-in_south [max_link_capacity=2]
