# Edge Case 1: Restricted Funnel Hazard
# Tests destination capacity reservation on high-throughput connections feeding into restricted zones.
# The connection start-r_gate allows up to 4 drones at once, but r_gate only has max_drones=1 and is restricted (2 turns).
# Multiple drones must NEVER enter the connection simultaneously if doing so would exceed r_gate's capacity on landing.
# An alternate detour path exists with equal hop count to allow optimal route splitting.

nb_drones: 6

start_hub: start 0 0 [color=green]
hub: r_gate 1 1 [zone=restricted color=red max_drones=1]
hub: r_exit 2 1 [zone=normal color=blue max_drones=2]
hub: detour1 1 -1 [zone=normal color=cyan max_drones=2]
hub: detour2 2 -1 [zone=normal color=cyan max_drones=2]
end_hub: goal 3 0 [color=green]

# High capacity connection into restricted gate (the hazard)
connection: start-r_gate [max_link_capacity=4]
connection: r_gate-r_exit [max_link_capacity=2]
connection: r_exit-goal [max_link_capacity=2]

# Balanced detour corridor
connection: start-detour1 [max_link_capacity=2]
connection: detour1-detour2 [max_link_capacity=2]
connection: detour2-goal [max_link_capacity=2]
